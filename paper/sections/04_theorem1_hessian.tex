\section{Theorem 1: The Hessian Capacity Bound}
\label{sec:thm1}

This section establishes our first main result: the condition number of
the joint Gauss-Newton matrix grows with the ratio of module capacities.
The result is the geometric foundation for the dynamical analysis of
Section~\ref{sec:thm2}: an ill-conditioned loss landscape induces a
forced timescale separation between the modules, which Theorem~\ref{thm:twoscale}
then shows leads to gradient starvation.

\subsection{Main result}

\begin{theorem}[Hessian Capacity Bound]
\label{thm:hessian}
Let $G = J^\top J$ denote the Gauss-Newton matrix of the loss
\eqref{eq:loss} at a parameter $(\thetap, \phip)$. Under
Assumptions~\ref{ass:linear}--\ref{ass:gap}, there exist constants
$c_1, c_2 > 0$ depending only on the data covariance, the bottleneck
dimension $K$, and the model class of $g_\phip$ such that
\[
    \kappa_{\mathrm{eff}}(G)
    \;\;\geq\;\; c_1 \cdot \sqrt{\frac{\Cg}{\Cf}}
    \;-\; c_2.
\]
Equivalently, if $g_\phip$ has fixed depth and characteristic width
$M$ (so $\Cg = \Theta(M^2)$ for typical MLP, CNN, ViT architectures),
\[
    \kappa_{\mathrm{eff}}(G)
    \;\;\geq\;\; \tilde{c}_1 \cdot \frac{M}{\sqrt{\Cf}}
    \;-\; c_2.
\]
\end{theorem}

The proof decomposes the condition number bound into separate bounds on
the largest and smallest eigenvalues of $G$. We state the three
supporting lemmas before presenting the proof.

\subsection{Supporting lemmas}

\begin{lemma}[Spatial block scaling]
\label{lem:phi_scaling}
Under Assumptions~\ref{ass:reg}--\ref{ass:subg}, let $M$ denote the
characteristic width of $g_\phip$ (channel count for a CNN,
embedding dimension for a ViT, hidden layer width for an MLP). Then
the largest eigenvalue of the spatial Gauss-Newton block satisfies
\[
    \lambda_{\max}(J_\phip^\top J_\phip)
    \;=\; \Omega(M)
    \qquad \text{in expectation over random initialization.}
\]
For fixed-depth $g_\phip$, $\Cg = \Theta(M^2)$, so equivalently
\[
    \lambda_{\max}(J_\phip^\top J_\phip)
    \;=\; \Omega\!\left( \sqrt{\Cg} \right).
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:phi_scaling} is a direct application of the mean-field
analysis of Karakida et al.~\cite{karakida2019universal}, who derive
closed-form expressions for the leading eigenvalue of the Fisher
Information Matrix of fully connected and convolutional networks with
random Gaussian initialization. The scaling is linear in the parameter
count when widths and depth are matched. We verify empirically in
Section~\ref{sec:experiments} that the scaling persists for the
architectures we consider (linear, CNN, ViT).
\end{remark}

\begin{lemma}[Eigenvalue interlacing for block matrices]
\label{lem:interlacing}
For any symmetric block matrix $M = \begin{pmatrix} A & B \\ B^\top & D
\end{pmatrix}$ with principal blocks $A$ and $D$,
\[
    \lambda_{\max}(M) \geq \max\{\lambda_{\max}(A), \lambda_{\max}(D)\}.
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:interlacing} is a standard consequence of Cauchy's
interlacing theorem applied to principal submatrices
(see e.g.\ Bhatia~\cite{bhatia1997matrix}, Corollary III.1.5).
\end{remark}

\begin{lemma}[Residual-dimension rank bound and smallest-eigenvalue cap]
\label{lem:schur}
Let $J_\thetap \in \mathbb{R}^{N_{\mathrm{data}} HW N_{\mathrm{cls}}
\times \Cf}$ denote the Jacobian of the softmax residual $r =
\mathrm{softmax}(\hat{y}) - y_{\mathrm{onehot}}$ with respect to the
spectral parameters $\thetap$, aggregated over $N_{\mathrm{data}}$
training examples and all $HW$ pixels. Under
Assumptions~\ref{ass:linear}--\ref{ass:inputlip}:

\emph{(i) Rank bound.} The Gauss-Newton block of the spectral module
has rank bounded by
\[
    \mathrm{rank}(J_\thetap^\top J_\thetap)
    \;\leq\; \min\!\left( \Cf,\;\;
        N_{\mathrm{data}} \cdot HW \cdot (N_{\mathrm{cls}} - 1)
    \right).
\]
In the over-parameterised regime $\Cf > N_{\mathrm{data}} HW
(N_{\mathrm{cls}} - 1)$, the block is rank-deficient with at least
$\Cf - N_{\mathrm{data}} HW (N_{\mathrm{cls}} - 1)$ eigenvalues
exactly equal to zero.

\emph{(ii) Smallest positive eigenvalue cap.} There exists a
constant $C_2 = C_2(\widetilde{L}, \lambda_{\max}(\Sigma_X),
N_{\mathrm{cls}})$ independent of $\Cg$ (but depending on the depth
of $g_\phip$ through $\widetilde{L}$) such that
\[
    \lambda_{\min}^{+}\!\left( J_\thetap^\top J_\thetap \right)
    \;\leq\; \frac{C_2}{N_{\mathrm{data}} HW (N_{\mathrm{cls}} - 1)}.
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:schur}(i) is the per-pixel bottleneck extension of the
classical Fisher-rank bound for softmax classification, going back to
Amari~\cite{amari2016information} (rank-($N_{\mathrm{cls}} - 1$)
Fisher of the categorical family); the sample-level $N(N_{\mathrm{cls}}
- 1)$ formulation appears in Botev, Ritter, and
Barber~\cite{botev2017practical}, and the empirical spectrum
observation is due to Sagun et al.~\cite{sagun2017empirical} and
Papyan~\cite{papyan2020traces}. Our contribution is the extension to
compositional networks with a linear per-pixel bottleneck and
arbitrary spatial classifier, plus the explicit $\Cg$-independence
statement in part (ii).
\end{remark}

\begin{remark}[Gauss-Newton vs.\ full Hessian]
Lemma~\ref{lem:schur} bounds the Gauss-Newton block $J_\thetap^\top
J_\thetap$ (equivalently, the empirical Fisher). The full Hessian
carries an additional residual-weighted second-derivative term $\sum_i
r_i \cdot \partial^2 \ell_i / \partial \thetap^2$ that vanishes at
convergence (Sagun et al.~\cite{sagun2017empirical}). The bound
transfers to the full Hessian at any stationary point.
\end{remark}

\begin{remark}
The Schur complement bound is classical; see Horn and
Johnson~\cite{horn2013matrix}, Theorem 1.20. The second statement
uses the linearity of $f_\thetap$ to derive the Kronecker
factorisation $J_\thetap^\top J_\thetap = A^\top A \otimes \Sigma_X$
(under a common-linearisation assumption on $A = \partial \hat{y} /
\partial Z$); combined with the Bhatia spectral identity for
Kronecker products, this yields the explicit bound
$\lambda_{\min}^{+}(J_\thetap^\top J_\thetap) \leq L^2 \cdot
\lambda_{\min}(\Sigma_X)$ where $L$ is the classifier's gradient
Lipschitz constant. See Section~\ref{sec:proof_thm1} for the full
derivation.
\end{remark}

\subsection{Proof of Theorem~\ref{thm:hessian}}

\begin{proof}[Proof of Theorem~\ref{thm:hessian}]
The Gauss-Newton matrix admits the block decomposition
\[
    G = \begin{pmatrix}
        J_\thetap^\top J_\thetap & J_\thetap^\top J_\phip \\
        J_\phip^\top J_\thetap & J_\phip^\top J_\phip
    \end{pmatrix}.
\]
By Lemmas~\ref{lem:phi_scaling} and \ref{lem:interlacing},
\[
    \lambda_{\max}(G)
    \;\geq\; \lambda_{\max}(J_\phip^\top J_\phip)
    \;=\; \Omega\!\left( \sqrt{\Cg} \right),
\]
using $\Cg = \Theta(M^2)$ for fixed-depth architectures.
By Lemma~\ref{lem:schur} (see Section~\ref{sec:proof_thm1} for the
full derivation), the smallest nonzero eigenvalue of the spectral
Gauss-Newton block is capped by a $\Cg$-independent constant $C_2$
depending only on the data covariance and the classifier's Lipschitz
constant. Transfer to the full matrix via a range-space argument
(Section~\ref{sec:proof_thm1}) gives
\[
    \lambda_{\min}^{+}(G) \;\leq\; C_2.
\]
Combining with the $\lambda_{\max}$ bound and identifying
$\Cf = SK$ (Assumption~\ref{ass:linear}),
\[
    \kappa_{\mathrm{eff}}(G)
    \;=\; \frac{\lambda_{\max}(G)}{\lambda_{\min}^{+}(G)}
    \;\geq\; \frac{c_1' \cdot \sqrt{\Cg}}{C_2}
    \;=\; c_1 \cdot \sqrt{\frac{\Cg}{\Cf}}
\]
with $c_1 := c_1' \cdot \sqrt{\Cf} / C_2$ depending only on the data
covariance, the bottleneck dimension $K$, the input spectral dimension
$S$, and the model class of $g_\phip$; a constant $c_2 \geq 0$ absorbs
finite-$\Cg$ corrections from Karakida's concentration bound. This
completes the proof.
\end{proof}

\subsection{Implications}

Theorem~\ref{thm:hessian} establishes that the loss landscape is
\emph{geometrically pathological} whenever $\Cg \gg \Cf$. Two immediate
consequences follow.

\paragraph{Bounded learning rate forces timescale separation.}
The maximum stable learning rate for first-order optimization is
bounded above by $2 / \lambda_{\max}(G)$. Under
Theorem~\ref{thm:hessian}, this bound scales as $\Omega(1/\sqrt{\Cg})$. With
such a small learning rate, updates to $\thetap$---whose Hessian block
has smallest nonzero eigenvalue bounded above by a $\Cg$-independent
constant---progress at a rate that is $\Omega(\sqrt{\Cg/\Cf})$-fold
slower than updates to $\phip$. This is the dynamical regime analyzed
in Section~\ref{sec:thm2}.

\paragraph{No remedy from optimizer tuning.}
The condition number bound depends only on architectural parameters,
not on the optimizer or schedule. Adaptive methods (Adam, RMSProp)
rescale gradients but cannot remove the underlying curvature
asymmetry; they redistribute the timescale gap rather than eliminating
it. We discuss this further in Section~\ref{sec:discussion}.

\subsection{Comments on tightness}

The bound in Theorem~\ref{thm:hessian} is a \emph{lower bound} on the
condition number. In practice the empirical condition number can be
substantially larger because $\lambda_{\min}(G)$ is often dominated by
near-degenerate directions in the spectral block that the bound does
not capture. Experiment~1.1 in Section~\ref{sec:experiments} measures
$\kappa(G)$ empirically across a range of capacity ratios and confirms
that the predicted scaling holds with a slope close to the theoretical
prediction.

\begin{remark}[Defense against linear triviality]
\label{rem:not_pca}
A natural objection is that, since $f_\thetap$ is linear, the analysis
must reduce to principal component analysis (PCA) of the input. This
is not the case. PCA finds directions of maximum input variance
\emph{without reference to labels}. Theorem~\ref{thm:hessian} concerns
the curvature of the supervised cross-entropy loss, which depends on
the current state of $g_\phip$ through the Jacobian
$\partial \hat{y}/\partial Z$. The pathology we identify is precisely
that this supervised signal fails to reach $\thetap$, not that
$f_\thetap$ is incapable of representing useful features.
\end{remark}
