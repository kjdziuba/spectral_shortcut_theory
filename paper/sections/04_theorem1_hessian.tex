\section{Theorem 1: The Hessian Capacity Bound}
\label{sec:thm1}

This section establishes our first main result: the condition number of
the joint Gauss-Newton matrix grows with the ratio of module capacities.
The result is the geometric foundation for the dynamical analysis of
Section~\ref{sec:thm2}: an ill-conditioned loss landscape induces a
forced timescale separation between the modules, which Theorem~\ref{thm:twoscale}
then shows leads to gradient starvation.

\subsection{Main result}

\begin{theorem}[Capacity-Induced Module Curvature Disparity]
\label{thm:hessian}
Consider the compositional model $F_{\thetap,\phip} = g_\phip \circ
f_\thetap$ with fixed-dimensional linear spectral module $f_\thetap$
(Assumption~\ref{ass:linear}) and a family of spatial networks
$g_\phip^{(M)}$ indexed by characteristic width $M$
(channels for a CNN, embedding dimension for a ViT, hidden width for
an MLP). Under
Assumptions~\ref{ass:linear}--\ref{ass:inputlip}, there exist
constants $c_1, c_2 > 0$ depending on the data covariance, the
bottleneck dimension $K$, the input spectral dimension $S$, and the
depth of $g_\phip$---but not on $M$ or on the spatial parameter count
$\Cg$---such that
\[
    \mathcal{D}_{\mathrm{curv}}^{(M)}
    \;=\; \frac{\lambda_{\max}(J_\phip^{(M)\top} J_\phip^{(M)})}
                {\lambda_{\max}(J_\thetap^\top J_\thetap)}
    \;\geq\; c_1 \cdot M - c_2
\]
in expectation over random Gaussian initialization. Equivalently, for
fixed-depth architectures with $\Cg = \Theta(M^2)$,
\[
    \mathcal{D}_{\mathrm{curv}}^{(M)}
    \;\geq\; c_1' \cdot \sqrt{\Cg / \Cf} - c_2.
\]
\end{theorem}

\begin{remark}[Interpretation]
Theorem~\ref{thm:hessian} states that as the spatial network widens,
its top Gauss-Newton eigenvalue grows without bound while the
spectral module's top Gauss-Newton eigenvalue remains capped by a
$\Cg$-independent constant. Curvature increasingly concentrates in
the downstream module. This is a purely geometric fact about the
Hessian; the dynamical consequence---that joint training pins
$\thetap$ near its initialization---is stated separately in
Theorem~\ref{thm:twoscale}, under its own hypotheses.
\end{remark}

The proof decomposes the condition number bound into separate bounds on
the largest and smallest eigenvalues of $G$. We state the three
supporting lemmas before presenting the proof.

\subsection{Supporting lemmas}

\begin{lemma}[Spatial block scaling]
\label{lem:phi_scaling}
Under Assumptions~\ref{ass:reg}--\ref{ass:subg}, let $M$ denote the
characteristic width of $g_\phip$ (channel count for a CNN,
embedding dimension for a ViT, hidden layer width for an MLP). Then
the largest eigenvalue of the spatial Gauss-Newton block satisfies
\[
    \lambda_{\max}(J_\phip^\top J_\phip)
    \;=\; \Omega(M)
    \qquad \text{in expectation over random initialization.}
\]
For fixed-depth $g_\phip$, $\Cg = \Theta(M^2)$, so equivalently
\[
    \lambda_{\max}(J_\phip^\top J_\phip)
    \;=\; \Omega\!\left( \sqrt{\Cg} \right).
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:phi_scaling} is a direct application of the mean-field
analysis of Karakida et al.~\cite{karakida2019universal}, who derive
closed-form expressions for the leading eigenvalue of the Fisher
Information Matrix of fully connected and convolutional networks with
random Gaussian initialization. The scaling is linear in the parameter
count when widths and depth are matched. We verify empirically in
Section~\ref{sec:experiments} that the scaling persists for the
architectures we consider (linear, CNN, ViT).
\end{remark}

\begin{lemma}[Eigenvalue interlacing for block matrices]
\label{lem:interlacing}
For any symmetric block matrix $M = \begin{pmatrix} A & B \\ B^\top & D
\end{pmatrix}$ with principal blocks $A$ and $D$,
\[
    \lambda_{\max}(M) \geq \max\{\lambda_{\max}(A), \lambda_{\max}(D)\}.
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:interlacing} is a standard consequence of Cauchy's
interlacing theorem applied to principal submatrices
(see e.g.\ Bhatia~\cite{bhatia1997matrix}, Corollary III.1.5).
\end{remark}

\begin{lemma}[Spectral block operator-norm cap]
\label{lem:schur}
Under Assumptions~\ref{ass:linear} (linear per-pixel $f_\thetap$)
and~\ref{ass:inputlip} (bounded input Jacobian
$\|\partial g_\phip / \partial Z\|_{\mathrm{op}} \leq \widetilde{L}$
uniformly in $\Cg$), the spectral Gauss-Newton block satisfies
\[
    \lambda_{\max}(J_\thetap^\top J_\thetap)
    \;=\; \|J_\thetap\|_{\mathrm{op}}^{2}
    \;\leq\; \widetilde{L}^{2} \cdot K \cdot \lambda_{\max}(\Sigma_X)
    \;=:\; C_\thetap,
\]
where $C_\thetap$ depends on the classifier's input Lipschitz
constant $\widetilde{L}$, the bottleneck dimension $K$, and the top
eigenvalue of the per-pixel input covariance $\Sigma_X$---but not on
the spatial width $M$ or the spatial parameter count $\Cg$.
\end{lemma}

\begin{remark}
Lemma~\ref{lem:schur} follows from the chain-rule identity
$J_\thetap = J_Z \cdot (\partial Z / \partial \thetap)$ combined with
$\|J_Z\|_{\mathrm{op}} \leq \widetilde{L}$
(Assumption~\ref{ass:inputlip}) and the exact form of $\partial Z /
\partial \thetap$ under linearity of $f_\thetap$
(Assumption~\ref{ass:linear}, giving $\partial Z / \partial \thetap =
I_K \otimes X^\top$ with $\|\partial Z / \partial
\thetap\|_{\mathrm{op}}^{2} \leq K \cdot \lambda_{\max}(\Sigma_X)$).
The bound is on the operator norm (equivalently, the top eigenvalue),
which is well-behaved under rank-deficiency of the block---avoiding
the smallest-eigenvalue and Kronecker-factorisation machinery of
earlier drafts. See Section~\ref{sec:proof_thm1} for details.
\end{remark}

\subsection{Proof of Theorem~\ref{thm:hessian}}

\begin{proof}[Proof of Theorem~\ref{thm:hessian}]
By Lemma~\ref{lem:phi_scaling}, in expectation over random Gaussian
initialization,
\[
    \lambda_{\max}(J_\phip^{(M)\top} J_\phip^{(M)}) \;\geq\; c_\phi \cdot M
\]
for a constant $c_\phi > 0$ depending only on the depth of $g_\phip$
and the data covariance, with concentration in the wide-network limit
(Karakida et al.~\cite{karakida2019universal}, Theorem~5.1). By
Lemma~\ref{lem:schur},
\[
    \lambda_{\max}(J_\thetap^\top J_\thetap) \;\leq\; C_\thetap
\]
for a constant $C_\thetap > 0$ independent of $M$. Dividing,
\[
    \mathcal{D}_{\mathrm{curv}}^{(M)}
    \;=\; \frac{\lambda_{\max}(J_\phip^{(M)\top} J_\phip^{(M)})}
                {\lambda_{\max}(J_\thetap^\top J_\thetap)}
    \;\geq\; \frac{c_\phi \cdot M}{C_\thetap}
    \;=\; c_1 \cdot M
\]
with $c_1 := c_\phi / C_\thetap$. A constant $c_2 \geq 0$ absorbs
finite-$M$ corrections from Karakida's concentration bound. Under
$\Cg = \Theta(M^2)$ (fixed-depth architectures), $M = \Theta(\sqrt{\Cg})$
and the ratio form $c_1' \cdot \sqrt{\Cg / \Cf}$ follows on multiplying
and dividing by $\sqrt{\Cf}$. This completes the proof.
\end{proof}

\subsection{Implications}

Theorem~\ref{thm:hessian} establishes that curvature concentrates in
the downstream spatial module as its width grows. We highlight two
implications, and separate them explicitly from the dynamical
consequences of Section~\ref{sec:thm2}.

\paragraph{Geometric mechanism, not a dynamical claim.}
$\mathcal{D}_{\mathrm{curv}} \gg 1$ identifies a
\emph{potential} for disparate module learning rates: the spatial
Gauss-Newton block admits directions of arbitrarily large curvature
while the spectral block does not. Whether joint gradient descent
actually produces disparate effective learning rates depends on
additional dynamical hypotheses (e.g., that the spatial module
converges quickly on the fast manifold), which we introduce and
verify in Section~\ref{sec:thm2}. Theorem~\ref{thm:hessian} does not,
by itself, prove that $\thetap$ is pinned to initialization.

\paragraph{No remedy from optimizer tuning of the shared learning rate.}
The disparity bound depends only on architectural parameters, not on
the choice of optimizer or the shared step-size schedule. Adaptive
methods (Adam, RMSProp) rescale per-coordinate gradient magnitudes
but do not alter the underlying eigenvalue disparity between the two
blocks. Two-timescale optimizers with explicitly different learning
rates per module (Heusel et al.~\cite{heusel2017gans}; Borkar
2008~\cite{borkar2008stochastic} Ch.~6) are the natural remedy at the
optimizer level. We discuss this in Section~\ref{sec:discussion}.

\begin{remark}[Small eigenvalues: irrelevance vs.\ starvation]
\label{rem:irrelevance_vs_starvation}
A parameter direction can carry vanishing curvature for two very
different reasons: (a) the direction genuinely does not affect
predictions on the task at hand (``irrelevance''), in which case zero
gradient is the correct outcome and the parameter should not be
learned; or (b) the direction \emph{would} affect predictions if the
optimizer could reach it, but the compositional geometry buries the
gradient signal behind the bottleneck (``starvation''). The
Gauss-Newton bounds of Theorem~\ref{thm:hessian} alone cannot
distinguish between these interpretations: both produce
$\Cg$-independent small curvature in the spectral block.

The synthetic experiments in Section~\ref{sec:experiments} are
designed to distinguish these hypotheses by construction. In
Experiment~1.3 we generate data where each of the two modalities alone
carries information sufficient to solve the task (single-modality
baselines reach $\approx 0.53$ on a binary classification, above the
$0.50$ chance floor). Joint training on the same data does not merely
under-use the spectral channel: it undershoots the weaker of the two
single-modality baselines. The gap is modest in absolute terms on this
synthetic problem---an important honest caveat---but its direction is
inconsistent with interpretation~(a), under which joint training
would at worst match the stronger single-modality baseline.

For real spectroscopic data, distinguishing (a) from (b) requires
demonstrating that a frozen pretrained spectral encoder outperforms
one that has been jointly trained end-to-end on the same task. We
report such experiments on FTIR and QCL tissue classification in
Section~\ref{sec:real_data}. Our present study compares only two
concrete mitigation strategies (freezing the spectral module, and
Pezeshki's Spectral
Decoupling~\cite{pezeshki2021gradient}); broader comparisons against
modality-specific learning rates, gradient surgery, GradNorm-style
loss balancing, and related interventions are important future work.
\end{remark}

\subsection{Empirical support}

Experiment~1.1 in Section~\ref{sec:experiments} measures
$\mathcal{D}_{\mathrm{curv}}^{(M)}$ directly across a range of spatial
widths and confirms monotonic growth with $M$. The observed
log-log slope $\approx 0.70$ against $\Cg$ falls between the
asymptotic prediction of Lemma~\ref{lem:phi_scaling} ($0.5$) and the
naive linear-in-parameters prediction ($1.0$), consistent with a
finite-width upward correction of the same character reported in
Karakida's own numerical validation~\cite{karakida2019universal}.

\begin{remark}[Defense against linear triviality]
\label{rem:not_pca}
A natural objection is that, since $f_\thetap$ is linear, the analysis
must reduce to principal component analysis (PCA) of the input. This
is not the case. PCA finds directions of maximum input variance
\emph{without reference to labels}. Theorem~\ref{thm:hessian} concerns
the curvature of the supervised cross-entropy loss, which depends on
the current state of $g_\phip$ through the Jacobian
$\partial \hat{y}/\partial Z$. The pathology we identify is precisely
that this supervised signal fails to reach $\thetap$, not that
$f_\thetap$ is incapable of representing useful features.
\end{remark}
